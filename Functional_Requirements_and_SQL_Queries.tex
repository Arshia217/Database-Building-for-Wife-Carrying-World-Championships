\begin{itemize}[leftmargin=1.5em, itemsep=1.5pt, topsep=1pt]
  \item \textbf{DBMS Engine:} MariaDB 11.8.8 / MySQL 8.0 (InnoDB engine, \texttt{utf8mb4} character set).

\section{Query (Read) Operations Implementation}

\subsection{Feature R1: Approved Entries per Edition (Read Operation 1)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{GET /entries} \quad$\vert$\quad \textbf{Function:} \texttt{query\_approved\_entries(edition\_year, category\_id)}
  \item \textbf{Description:} Filters confirmed competitors by edition and category. Joins 6 tables using \texttt{CONCAT\_WS} for participant names.
\end{itemize}

\begin{lstlisting}
SELECT 
    ce.entry_id, ce.bib_number, ce.edition_year, ce.registration_status, ce.registration_time,
    c.category_name, pe.carrier_id,
    CONCAT_WS(' ', p1.first_name, p1.last_name) AS carrier_name, co1.country_name AS carrier_country,
    pe.carried_id,
    CONCAT_WS(' ', p2.first_name, p2.last_name) AS carried_name, co2.country_name AS carried_country,
    te.team_name
FROM competition_entry ce
JOIN category c ON ce.category_id = c.category_id
LEFT JOIN pair_entry pe ON ce.entry_id = pe.entry_id
LEFT JOIN participant p1 ON pe.carrier_id = p1.participant_id
LEFT JOIN country co1 ON p1.country_code = co1.country_code
LEFT JOIN participant p2 ON pe.carried_id = p2.participant_id
LEFT JOIN country co2 ON p2.country_code = co2.country_code
LEFT JOIN team_entry te ON ce.entry_id = te.entry_id
WHERE ce.edition_year = %s AND ce.registration_status = 'Approved'
  AND (%s IS NULL OR ce.category_id = %s)
ORDER BY ce.bib_number IS NULL, ce.bib_number, ce.entry_id;
\end{lstlisting}

\subsection{Feature R2: Participant Name Search (Read Operation 2)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{GET /search} \quad$\vert$\quad \textbf{Function:} \texttt{query\_name\_search(name\_fragment)}
  \item \textbf{Description:} Wildcard pattern matching (\texttt{LIKE \%s}) across full names, displaying nationality, editions, categories, and roles.
\end{itemize}

\begin{lstlisting}
SELECT 
    p.participant_id, CONCAT_WS(' ', p.first_name, p.last_name) AS full_name,
    p.first_name, p.last_name, p.approval_status, co.country_name,
    ce.entry_id, ce.edition_year, ce.bib_number, c.category_name,
    pi.participant_role, ce.registration_status
FROM participant p
JOIN country co ON p.country_code = co.country_code
LEFT JOIN participates_in pi ON p.participant_id = pi.participant_id
LEFT JOIN competition_entry ce ON pi.entry_id = ce.entry_id
LEFT JOIN category c ON ce.category_id = c.category_id
WHERE CONCAT_WS(' ', p.first_name, p.last_name) LIKE %s
ORDER BY p.participant_id, ce.edition_year DESC;
\end{lstlisting}

\subsection{Feature R3: Entries per Category Breakdown (Read Operation 3)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{GET /categories} \quad$\vert$\quad \textbf{Function:} \texttt{query\_entries\_per\_category(edition\_year)}
  \item \textbf{Description:} Aggregates total, approved, and pending registration volume per category for a given edition using conditional sums.
\end{itemize}

\begin{lstlisting}
SELECT 
    hc.edition_year, c.category_id, c.category_name,
    COUNT(ce.entry_id) AS total_entries,
    COALESCE(SUM(CASE WHEN ce.registration_status = 'Approved' THEN 1 ELSE 0 END), 0) AS approved_entries,
    COALESCE(SUM(CASE WHEN ce.registration_status = 'Pending' THEN 1 ELSE 0 END), 0) AS pending_entries
FROM has_category hc
JOIN category c ON hc.category_id = c.category_id
LEFT JOIN competition_entry ce 
    ON hc.edition_year = ce.edition_year AND hc.category_id = ce.category_id
WHERE (%s IS NULL OR hc.edition_year = %s)
GROUP BY hc.edition_year, c.category_id, c.category_name
ORDER BY hc.edition_year DESC, c.category_id;
\end{lstlisting}

\subsection{Feature R4: Heat Results \& Time Differences (Read Operation 4)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{GET /results}
  \item \textbf{Function:} \texttt{query\_heat\_result\_sheet(edition\_year, race\_no, heat\_no)} in \texttt{src/services.py}
  \item \textbf{Description:} Computes final time (\texttt{raw\_time + penalty\_seconds}) and delta to winner via the window function \texttt{MIN(...) OVER ()}. Handles nullable officials with \texttt{LEFT JOIN}.
\end{itemize}

\begin{lstlisting}
SELECT 
    rr.edition_year, rr.race_no, rr.heat_no, rr.`rank`, rr.result_status,
    ce.entry_id, ce.bib_number, te.team_name,
    CONCAT_WS(' ', p1.first_name, p1.last_name) AS carrier_name,
    CONCAT_WS(' ', p2.first_name, p2.last_name) AS carried_name,
    rr.raw_time, rr.penalty_seconds,
    CASE WHEN rr.result_status = 'Finished' THEN (rr.raw_time + rr.penalty_seconds) ELSE NULL END AS final_time,
    CASE 
        WHEN rr.result_status = 'Finished' THEN 
            ROUND((rr.raw_time + rr.penalty_seconds) - 
                  MIN(CASE WHEN rr.result_status = 'Finished' THEN (rr.raw_time + rr.penalty_seconds) END) OVER (), 3)
        ELSE NULL 
    END AS time_diff,
    CONCAT_WS(' ', o.first_name, o.last_name) AS official_name
FROM records_result rr
JOIN competition_entry ce ON rr.entry_id = ce.entry_id
LEFT JOIN pair_entry pe ON ce.entry_id = pe.entry_id
LEFT JOIN participant p1 ON pe.carrier_id = p1.participant_id
LEFT JOIN participant p2 ON pe.carried_id = p2.participant_id
LEFT JOIN team_entry te ON ce.entry_id = te.entry_id
LEFT JOIN official o ON rr.official_id = o.official_id
WHERE rr.edition_year = %s AND rr.race_no = %s AND rr.heat_no = %s
ORDER BY 
    CASE WHEN rr.result_status = 'Finished' THEN 0 ELSE 1 END,
    rr.`rank` IS NULL, rr.`rank`, final_time;
\end{lstlisting}

\subsection{Feature R5: Category Average Finish Times (Read Operation 5)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{GET /categories\#averages} \quad$\vert$\quad \textbf{Function:} \texttt{query\_average\_final\_times(edition\_year)}
  \item \textbf{Description:} Performs statistical aggregation across competition categories for finished entries, computing finisher count, average final time, fastest time, and slowest time.
\end{itemize}

\begin{lstlisting}
SELECT 
    c.category_id, c.category_name,
    COUNT(rr.entry_id) AS total_finishes,
    ROUND(AVG(rr.raw_time + rr.penalty_seconds), 3) AS avg_final_time,
    ROUND(MIN(rr.raw_time + rr.penalty_seconds), 3) AS fastest_final_time,
    ROUND(MAX(rr.raw_time + rr.penalty_seconds), 3) AS slowest_final_time
FROM records_result rr
JOIN competition_entry ce ON rr.entry_id = ce.entry_id
JOIN category c ON ce.category_id = c.category_id
WHERE rr.result_status = 'Finished'
  AND (%s IS NULL OR rr.edition_year = %s)
GROUP BY c.category_id, c.category_name
ORDER BY avg_final_time ASC;
\end{lstlisting}

\section{Update (Write) Operations Implementation}

\subsection{Feature W1: Register a Pair (Write Operation 1 --- Create)}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Route:} \texttt{POST /register} \quad$\vert$\quad \textbf{Function:} \texttt{write\_register\_pair(...)} in \texttt{src/services.py}
  \item \textbf{Description:} Multi-table atomic transaction taking text inputs for carrier and carried competitors. Public entries default to \texttt{'Pending'} (no option to choose status); authenticated admins may optionally choose status. Automatically inserts new competitor records into \texttt{participant} with \texttt{approval\_status}.
\end{itemize}

\begin{lstlisting}
-- Step 1: Validate edition offers chosen category
SELECT 1 FROM has_category WHERE edition_year = %s AND category_id = %s;

-- Step 2: Validate country existence & check if competitor already exists in DB
SELECT country_code FROM country WHERE country_code = %s;
SELECT participant_id, first_name, last_name, date_of_birth, weight_kg, country_code, approval_status 
FROM participant 
WHERE LOWER(first_name) = LOWER(%s) 
  AND ((last_name IS NULL AND %s IS NULL) OR LOWER(last_name) = LOWER(%s))
  AND country_code = %s LIMIT 1;

-- Step 3: Insert new competitor into participant table with approval_status
INSERT INTO participant (first_name, last_name, date_of_birth, weight_kg, country_code, approval_status)
VALUES (%s, %s, %s, %s, %s, %s);

-- Step 4: Insert master competition entry record
INSERT INTO competition_entry (edition_year, category_id, bib_number, registration_status)
VALUES (%s, %s, %s, %s);

-- Step 5: Insert pair_entry subtype specialization using generated entry_id
INSERT INTO pair_entry (entry_id, carrier_id, carried_id)
VALUES (%s, %s, %s);

-- Step 6: Insert Carrier and Carried roles into participates_in
INSERT INTO participates_in (entry_id, participant_id, participant_role)
VALUES (%s, %s, 'Carrier'), (%s, %s, 'Carried');

-- Step 7: Record participant-edition history in participates_in_edition
INSERT IGNORE INTO participates_in_edition (participant_id, edition_year)
VALUES (%s, %s), (%s, %s);
\end{lstlisting}

\subsection{Feature W3: Delete Pending Entry (Write Operation 3 --- Delete) \& Approval}
\begin{itemize}[leftmargin=1.5em, itemsep=0.5pt, topsep=0.5pt]
  \item \textbf{Routes:} \texttt{POST /entries/delete/<entry\_id>} \quad$\vert$\quad \texttt{POST /entries/approve/<entry\_id>}

        \hphantom{\textbf{Functions: }}\texttt{write\_approve\_pending\_entry(entry\_id)}
  \item \textbf{Description:} Admin-only write operation (protected by \texttt{@admin\_required}). Allows administrators to approve pending entries or safely delete them with edition cleanup and automatic orphan competitor removal.
\end{itemize}

\begin{lstlisting}
-- Step 1: Verify entry exists & fetch edition_year / participants
SELECT entry_id, edition_year, registration_status FROM competition_entry WHERE entry_id = %s;
SELECT participant_id FROM participates_in WHERE entry_id = %s;

-- Step 2: Reject deletion if heat results already exist; otherwise cascade clean-up
SELECT 1 FROM records_result WHERE entry_id = %s;
DELETE FROM participates_in WHERE entry_id = %s;
DELETE FROM pair_entry WHERE entry_id = %s;
DELETE FROM team_entry WHERE entry_id = %s;
DELETE FROM assigned_to WHERE entry_id = %s;
DELETE FROM competition_entry WHERE entry_id = %s;

-- Step 3: Remove edition participation & orphaned competitors with no remaining entries
DELETE FROM participates_in_edition WHERE participant_id = %s AND edition_year = %s;
DELETE FROM participant_contact WHERE participant_id = %s;
DELETE FROM participant WHERE participant_id = %s;

-- [Approval Workflow]: Confirm entry & promote participant approval_status
UPDATE competition_entry SET registration_status = 'Approved' WHERE entry_id = %s;
UPDATE participant SET approval_status = 'Approved' 
WHERE participant_id IN (SELECT participant_id FROM participates_in WHERE entry_id = %s);
\end{lstlisting}

\section{Database Integrity \& Security Architecture}

\begin{enumerate}[leftmargin=1.5em, itemsep=1.2pt, topsep=1pt]
  \item \textbf{ACID Transaction Management:} Multi-table write workflows (W1 and W3) utilize the Python context manager \texttt{db.transaction()} with \texttt{autocommit=False}. If any constraint, foreign key, or business check fails, an automatic rollback is immediately triggered.
  \item \textbf{Protection Against SQL Injection:} Every SQL statement executed across the Flask backend strictly uses parameter substitution with \texttt{\%s} placeholders. No manual string concatenation is used for user inputs.
  \item \textbf{Database-Level Disjoint Triggers:} The database schema enforces disjoint specialization natively via triggers (\texttt{trg\_pair\_not\_team} and \texttt{trg\_team\_not\_pair}), ensuring no entry can simultaneously exist in \texttt{pair\_entry} and \texttt{team\_entry}.
  \item \textbf{Safe Handling of Nullable Columns:} Left outer joins (\texttt{LEFT JOIN}) and \texttt{CONCAT\_WS} ensure that unrecorded attributes (e.g., unpublished officials or participants with single-word names) are handled gracefully without omitting rows or breaking page rendering.
  \item \textbf{Participant Lifecycle \& Role-Based Access Control:} Competitor text inputs dynamically set \texttt{approval\_status} (\texttt{'Pending'} or \texttt{'Approved'}). Deleting a pending entry purges \texttt{participates\_}\allowbreak\texttt{in\_edition} and deletes orphaned competitors. Race administrators authenticate via parameterized SQL against \texttt{admin\_user} with secure password hashing; public writes are strictly limited to pair registration (forced to \texttt{'Pending'}).
\end{enumerate}

All 8 features and underlying queries have been verified against the live MariaDB 11.8 database populated with the official Eukonkanto championship dataset:
\begin{itemize}[leftmargin=1.5em, itemsep=1pt, topsep=1pt]

\end{itemize}
